import java.util.Scanner;

class Adam {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int rev = 0, temp = n;

        while (temp > 0) {
            rev = rev * 10 + temp % 10;
            temp /= 10;
        }

        int a = n * n;
        int b = rev * rev;
        int r = 0;

        while (b > 0) {
            r = r * 10 + b % 10;
            b /= 10;
        }

        System.out.println(a == r ? "Adam" : "Not Adam");
    }
}
