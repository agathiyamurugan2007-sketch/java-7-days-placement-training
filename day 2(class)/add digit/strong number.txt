import java.util.Scanner;

class Strong {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt(), temp = n, sum = 0;

        while (n > 0) {
            int d = n % 10;
            int f = 1;

            for (int i = 1; i <= d; i++)
                f *= i;

            sum += f;
            n /= 10;
        }

        System.out.println(sum == temp ? "Strong" : "Not Strong");
    }
}
