public class HollowArrow {
    public static void main(String[] args) {
        int n = 5;

        // Arrow head
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++)
                System.out.print(" ");

            for (int j = 1; j <= 2 * i - 1; j++) {
                if (i == n || j == 1 || j == 2 * i - 1)
                    System.out.print("*");
                else
                    System.out.print(" ");
            }
            System.out.println();
        }

        // Arrow tail
        for (int i = 1; i <= n - 1; i++) {
            for (int j = 1; j < n; j++)
                System.out.print(" ");
            System.out.println("*");
        }
    }
}
