import java.util.*;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int t = sc.nextInt();

        while (t-- > 0) {
            int p = sc.nextInt(), count = 0;

            for (int i = 2048; i >= 1; i /= 2) {
                count += p / i;
                p %= i;
            }

            System.out.println(count);
        }
    }
}
